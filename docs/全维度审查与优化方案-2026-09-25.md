#### 因为要真动，这几条护栏反而更重要（人操作时的兜底）

真机可控 ⇒ **误触代价真实存在** ⇒ 下面几条从"锦上添花"升级为"必须"：

1. **`require_live_safety` 现场运行期建议置 `true`**（`robot.yaml:128`）——让服务端围栏互锁真正生效（当前 fail-open，形同虚设，见 B-14）。
2. **B-01 程序号白名单** —— 否则一次手误/一个旧脚本就能让控制器跑 411（B-01）。
3. **F-01 点动松手不停** —— 松手后机器人继续走，只能拍急停（F-01）。**这一条与"动哪根轴"无关，全轴可动时更致命。**
4. **B-03 速度回读比对** —— 你设多少，控制器实际是不是那个值，你得知道（B-03）。
5. **B-06 急停结果可见** —— 急停没送达时必须明确报错，而不是显示"已停"（B-06）。
6. **F-03（重新定位）** —— 把硬编码的 J1–J5 锁改成**可配置模式**：默认全轴可动 + 打开开关时仅 J6（§2.9）。

> 一句话：**轴的选择权归你，护栏由代码兜，AI 只在 J6 + 5% 的笼子里活动**。

```python
def estop(self) -> dict:
    """急停：停止当前运动 + 锁定引擎。★ 不排 _exec_lock（急停不能被排队）。"""
    with self._lock:
        self.stopped = True
        self.moving = False
```

### 2.7 F-01　点动「松手不停」竞态（最危险的 UI Bug）

**定位**：`frontend/src/stores/exec.js:444-460`（`jogPress/jogRelease`）、`:364-404`（`startJog`）、`:386-389`（keepAlive）
**现象**：按住 260ms 以上后松手，前端既发一次增量点动（`stepJog`）又把连续点动跑起来，并每 400ms 持续 keepalive → **松手后机器人持续运动**，只能靠限位/围栏/急停终止。
**触发**：按住时长落在 `[260ms, 260ms + RTT]` 区间（局域网几十毫秒，后端偶发慢响应极易命中）。
**根因**：① `pressTimer` 回调调 `startJog` 后不清空 `pressInfo`；② `startJog` 在 `await` 之后才置 `holding`，窗口内 `holding` 仍 −1；③ `jogRelease` 判定 `holding>=0` 失败 → 把已作废的 `pressInfo` 当短按；④ `startJog` 返回后不校验"用户是否仍按住"。

**修复代码**（三处）：

```js
// stores/exec.js —— 模块作用域新增"按压世代"令牌
let pressToken = 0;

jogPress(joint, dir) {
  if (this.needAuth()) return;
  if (this.UNSAFE) return;
  if (jointLocked(joint)) {                       // ★ 配合 §2.9
    // ★ v2.1：jointLocked 是「模式驱动」——默认模式下恒为 false（全轴可动）
    this.logLine("warn", `轴锁模式：J${joint} 不可动，仅 J${this.jointLock?.only || 6} 可动`);
    return;
  }
  const my = ++pressToken;                        // ★ 新一次按压
  pressInfo = { joint, dir, token: my };
  if (pressTimer) clearTimeout(pressTimer);
  pressTimer = setTimeout(() => {
    pressTimer = null;
    const info = pressInfo;
    pressInfo = null;                             // ★ 补上：起步即作废
    if (info) this.startJog(info.joint, info.dir, my);
  }, HOLD_MS);
},

jogRelease() {
  pressToken++;                                   // ★ 松手即作废在途的 startJog
  if (pressTimer) { clearTimeout(pressTimer); pressTimer = null; }
  const info = pressInfo;
  pressInfo = null;
  if (this.holding >= 0) { this.endJog(); return; }
  if (info) this.stepJog(info.joint, info.dir);
},

async startJog(joint, dir, token) {
  if (!useAuthStore().controlActive) return;
  if (jointLocked(joint)) return;
  this.jogBusy = true;
  try {
    const r = await apiControl("/control/jog/start", {
      method: "POST",
      body: JSON.stringify({ joint, dir, speed_pct: Number(this.jogSpeed) || 5 }),
    });
    const d = await r.json().catch(() => ({}));
    if (!r.ok) { this.logLine("err", "连续点动启动失败：" + (d.error || r.status)); return; }
    // ★ 关键：返回时用户可能已经松手，此时绝不能建定时器
    if (token !== undefined && token !== pressToken) {
      try { await apiControl("/control/jog/stop", { method: "POST" }); } catch (e) {}
      this.logLine("warn", "松手早于启动返回，已取消连续点动");
      return;
    }
    this.holding = joint;
    if (keepAliveTimer) clearInterval(keepAliveTimer);
    keepAliveTimer = setInterval(() => {
      if (this.holding < 0) return;
      apiControl("/control/jog/keepalive", { method: "POST" }).catch(() => {});
    }, JOG_KEEPALIVE_MS);
  } finally {
    this.jogBusy = false;
  }
}
```

**验收**：① 按住 300ms 松手 → 只产生一次 `jog/stop`，无 keepalive；② 按住 3s 松手 → `jog/stop` 后无后续请求；③ 弱网（DevTools 限速 3G）下重复 10 次，无"松手后仍在动"。

### 2.12 F-06　WebSocket 无心跳 → 半开连接永不重连

```js
const HB_MS = 3000;          // 3s 一跳
const HB_TIMEOUT_MS = 10000; // 10s 无回包视为死链

function startHeartbeat() {
  stopHeartbeat();
```

```python
         need_jog_mode: bool = True) -> Tuple[bool, str, dict]:
    """返回 (是否放行, 拒绝原因, 上下文)。拒绝原因可直接回给用户。"""
    ctx = {}
    # 1) 目标角合法性
    if not joints or len(joints) != 6:
        return False, "目标关节角缺失或长度不为 6，拒绝下发", ctx
    try:

    sim_mode = str(get_config().connection.get("simulate", "auto")).lower()

    if cur_p.get("stale"):

    # 4) 档位（B-13）
    if need_jog_mode:

        return False, "安全互锁未生效或当前处于危险区，已拒绝下发", ctx

    # 6) 轴锁（F-03 v2.1：可配置模式，默认关闭 → J1~J6 全轴可动）
    cur_j = [float(cur_p.get(f"j{i}", 0.0)) for i in range(1, 7)]
    if joint_lock_enabled():
        tol = joint_lock_tol()
        only = _only_joint()
        bad = []
        for i in locked_joints():
            dev = abs(js[i - 1] - cur_j[i - 1])
```

```python
from typing import List, Optional

def parse_joints(raw: Optional[str], *, where: str = "") -> List[float]:
    """★ 解析点位关节角。失败一律抛 ValueError，绝不返回全零。"""
    if raw is None or str(raw).strip() == "":

        except Exception:
            raise ValueError(f"{where}点位关节角含非数值：{x!r}")
        if not math.isfinite(f):
```

```python
# app/main.py —— 替换 BodyLimitMiddleware，改为在 ASGI 层真实计数
MAX_BODY_BYTES = int(os.getenv("EFORT_MAX_BODY", str(8 * 1024 * 1024)))

class BodyLimitMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request, call_next):
        cl = request.headers.get("content-length")
        if cl and cl.isdigit() and int(cl) > MAX_BODY_BYTES:
            return JSONResponse({"detail": "请求体过大"}, status_code=413)
        if request.headers.get("transfer-encoding", "").lower() == "chunked":
            total = 0
            real_receive = request.receive

            async def counting_receive():
                nonlocal total
                msg = await real_receive()
                if msg.get("type") == "http.request":
                    body = msg.get("body", b"") or b""
                    total += len(body)
                    if total > MAX_BODY_BYTES and not msg.get("more_body"):
                        pass
                    if total > MAX_BODY_BYTES:
                        raise HTTPException(status_code=413, detail="请求体过大")
                return msg
            request._receive = counting_receive
        return await call_next(request)
```

### 4.6 架构级功能：事件流单一订阅者（F-12）

```js
// frontend/src/services/eventsBus.js （新建）
import { wsEventsUrl } from "../config.js";
import { useAuthStore } from "../stores/auth.js";

const subs = new Set();
let ws = null, retry = 0, timer = null, pingTimer = null;
let lastPingAt = 0;

function emit(d) { subs.forEach((fn) => { try { fn(d); } catch (e) { console.error(e); } }); }

function open() {
  const auth = useAuthStore();
  if (!auth.token) return;
  ws = new WebSocket(wsEventsUrl());
  ws.onopen = () => {
    retry = 0;
    ws.send(JSON.stringify({ type: "auth", token: auth.token }));
    lastPingAt = Date.now();
    pingTimer = setInterval(() => {
      if (!ws || ws.readyState !== 1) return;
      if (Date.now() - lastPingAt > 15000) { try { ws.close(); } catch (e) {} return; }
      try { ws.send(JSON.stringify({ type: "ping", t: Date.now() })); } catch (e) {}
    }, 5000);
  };
  ws.onmessage = (ev) => {
    lastPingAt = Date.now();
    try { const d = JSON.parse(ev.data); if (d && d.type !== "pong") emit(d); } catch (e) {}
  };
  ws.onclose = () => {
    clearInterval(pingTimer);
    retry = Math.min(retry + 1, 6);
    timer = setTimeout(open, Math.min(500 * 2 ** retry, 8000) * (0.7 + Math.random() * 0.6));
  };
}

/** 订阅事件流；返回取消订阅函数。全站唯一 WebSocket。 */
export function subscribeEvents(fn) {
  subs.add(fn);
  if (!ws || ws.readyState !== 1) open();
  return () => {
    subs.delete(fn);
    if (subs.size === 0 && ws) {           // ★ 无人订阅时关闭，避免空转
      clearTimeout(timer); clearInterval(pingTimer);
      try { ws.close(); } catch (e) {}
      ws = null;
    }
  };
}
export function reconnectEvents() { try { ws && ws.close(); } catch (e) {} open(); }
```
