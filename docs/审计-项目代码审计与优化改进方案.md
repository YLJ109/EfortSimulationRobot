def encode_angle(deg: float) -> int:
    """角度 → 有符号 16 位（×100 补码）。量化 0.01°。超出范围直接拒绝。"""
    v = int(round(float(deg) * ANG_SCALE))

        raise ValueError("角度 %s° 超出 int16×100 表示范围(±327.67°)，拒绝下发" % deg)
    return v & 0xFFFF

def _echo_matches(self, got: int, want: int) -> bool:
    """回读比对必须按有符号值解码，不能比原始无符号字。"""
    def to_signed(w: int) -> int:
        w &= 0xFFFF
        return w - 0x10000 if w > 0x7FFF else w
    return to_signed(got) == to_signed(want)

## 4. 架构改进方案（工业监控场景专项）

### 4.1 通信层优化（机器人通信断线、心跳、重连）

**现状问题链**（P1-A2/A4/A5/A6 + P2 多项）：
读失败单次阻塞 2–4s → 降级耗时 60–120s（注释称 3s）→ 采集线程无兜底会死 →
半开连接靠 2s 超时兜、滴流响应可无限占 `_io_lock` → 无 TCP keepalive → 无重试。

**目标架构：**

```
┌─────────────────── ModbusLink（新建，唯一 socket 持有者）───────────────────┐
│  状态机: DISCONNECTED → CONNECTING → HEALTHY → DEGRADED → CONNECTED       │
│  · 连接: settimeout(1.0) + SO_KEEPALIVE(1,5,3)  ← 空闲 15s 探活           │
│  · 心跳: 独立低频线程每 3s 读 ADDR_STATUS(40001) 1 个寄存器                │
│          （成本极低，与采集解耦，用于"链路活着"判定）                      │
│  · 收帧: 整帧 deadline = timeout（防滴流占锁），length>300 判帧失步断线    │
│  · 重试: 首次失败 → _close() → 重建 socket → 重试一次（半开连接自愈）     │
│  · 退避: 断线探测 0.5s → 1 → 2 → 5s 封顶（替代 FAIL_LIMIT=30 的固定计数） │
│  · 帧解析: 全部纳入 try，BC/长度/奇偶校验，失败返回 (None, err)            │
│  · 锁: _io_lock 覆盖 IO + close（代际计数校验 socket 归属）                │
└────────────────────────────────────────────────────────────────────────────┘
         │ 心跳结果                      │ 读结果
         ▼                               ▼
   link store（前端底栏）          Collector（只读缓存，不阻塞）
```

**关键改造点（按优先级）：**

1. **拆出心跳线程**——把 `rc_snapshot`（3 个 FC3 事务、最坏 6s）从采集循环移出，
   采集循环只读缓存；心跳结果直接喂 `link` store，消除 `/system/guide` 的无缓存
   `_tcp_probe`（P1-B8）。
2. **失败即标注而非补帧**——采集失败沿用 `last_joints` 并置 `payload["stale"]=true`、
   `source="stale"`（P1-A6），前端据 `stale` 点亮黄色灯，杜绝伪造历史数据。
3. **执行序列原子化**——`_exec_lock` 覆盖"校验→写速度→写目标→回读→settle→触发"，
   并把速度写入纳入该序列（P0-2 + P1-A9）。
4. **触发位清理 finally 化** + 失败即写事件总线（P1-A1）。
5. **健康探针分级**：
   - L1 进程存活 `/healthz`（无 IO）
   - L2 链路存活（心跳线程结果，**无 HTTP 探测**）
   - L3 数据新鲜度（`last_pose_at`，前端 `stale` 判定）

# hub.py 改造骨架
class _Client:
    __slots__ = ("ws", "loop", "latest", "priority", "sending", "lock")

CRITICAL_TYPES = ("event", "rc_status", "estop")     # 关键帧永不被覆盖

def broadcast(self, payload: dict) -> None:
    data = json.dumps(payload, ensure_ascii=False, allow_nan=False)
    with self._lock:                                  # 保护 dict 迭代
        clients = list(self._clients.values())
    for c in clients:
        with c.lock:
            if payload.get("type") in CRITICAL_TYPES:
                if len(c.priority) >= 32:             # 有界，满则丢最旧的 info 级
                    c.priority.pop(0)
                c.priority.append(data)
            else:
                c.latest = data                       # pose 类覆盖式（背压）
            if c.sending:
                continue
            c.sending = True
    # 每个 client 的 drain 循环：先排空 priority，再取 latest
    # finally 中清 sending 后复查 latest/priority，消除丢唤醒窗口

| **L1 严重** | 速度被夹取、档位未确认放行、围栏过期 fail-open、写寄存器回读不一致 | 底栏黄灯 + 事件时间线 | 同上 |
| **L2 警告** | 数据 stale、断线降级模拟、登录失败、导入部分跳过 | 事件时间线 | 仅计数 |
| **L3 信息** | 连接恢复、配置变更、执行完成 | 事件时间线 | — |

**必须补的告警点（当前缺失）：**
1. **采集线程死亡** → 这是最危险的"假活"，需在心跳线程里检测
   `last_pose_at` 超时即报 L0（P1-A5）。
2. **入库连续失败** → `insert_pose` 失败 >10 次报 L1（当前只写一行日志，静默丢数据）。
3. **执行与请求不符** → 回读校验失败、触发位异常滞留报 L0。
4. **磁盘水位** → 启动与周期清理时检查 `data/`、`logs/` 剩余空间，<10% 报 L0。
5. **配置声明与实际不一致** → 启动自检发现 `apply="reconnect"` 项无法热更时打 WARN
   （P1-C2/C3/C4 的根因告警）。

**闭环**：所有 L0/L1 事件写入 `system_events` 并在运维页可确认（ack），
确认后不再重复播报；未确认的在底栏持续显示。
